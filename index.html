function recalcStreak(dates) {

  if (!dates.length) return 0;

  const sorted = [...dates].sort().reverse();

  const t = today(), y = new Date(Date.now()-86400000).toISOString().slice(0,10);

  if (sorted[0]!==t && sorted[0]!==y) return 0;

  if (tokens>=500) return {t:'👑 Легенда',    c:'#F59E0B'};

  if (tokens>=200) return {t:'💎 Мастер',     c:'#8B5CF6'};

  if (tokens>=100) return {t:'⚡ Продвинутый', c:'#3B82F6'};

  if (tokens>=50)  return {t:'🔥 Активный',   c:'#10B981'};

  return              {t:'🌱 Новичок',        c:'#6B7280'};

}

let toastTimer = null;

function showToast(msg, type='info') {

  clearTimeout(toastTimer);

  const old = document.querySelector('.toast');

  if (old) old.remove();

  const el = document.createElement('div');

  el.className = `toast ${type}`;

  el.textContent = msg;

  document.body.appendChild(el);

  toastTimer = setTimeout(() => el.remove(), 3000);

}


function makeAvt(name, size, extraCls='') {

  const div = document.createElement('div');

  div.className = `avt ${extraCls}`;

  div.style.cssText = `width:${size}px;height:${size}px;background:${avtBg(name)};font-size:${Math.round(size*.38)}px`;

  div.textContent = initials(name);

  return div;

}


function renderHabits() {

  const list = document.getElementById('habit-list');

  const empty = document.getElementById('empty-habits');

  const win = document.getElementById('prog-win');

  const done = ST.habits.filter(isDoneToday).length;

  const pct = ST.habits.length ? (done/ST.habits.length)*100 : 0;

  document.getElementById('prog-cnt').textContent = `${done} / ${ST.habits.length}`;

  document.getElementById('prog-fill').style.width = pct + '%';

  win.classList.toggle('hidden', !(done===ST.habits.length && ST.habits.length>0));

  list.innerHTML = '';

  if (!ST.habits.length) { empty.classList.remove('hidden'); return; }

  empty.classList.add('hidden');

  ST.habits.forEach(h => {

    const d = isDoneToday(h);

    const card = document.createElement('div');

    card.className = `hcard ${d?'done':''}`;

    const chk = document.createElement('button');

    chk.className = 'hchk';

    chk.style.borderColor = h.color;

    chk.style.background = d ? h.color : 'transparent';

    chk.textContent = d ? '✓' : '';

    chk.onclick = () => toggleHabit(h.id);

    const emo = document.createElement('div');

    emo.className = 'hemo';

    emo.style.background = h.color + '22';

    emo.textContent = h.emoji;

    const info = document.createElement('div');

    info.className = 'hinfo';

    const nm = document.createElement('span');

    nm.className = 'hname'; nm.textContent = h.name;

    info.appendChild(nm);

    if (h.streak > 0) {

      const st = document.createElement('span');

      st.className = 'hstrk'; st.textContent = `🔥 ${h.streak} д.`;

      info.appendChild(st);

    }

    const del = document.createElement('button');

    del.className = 'hdel'; del.textContent = '×';

    del.onclick = () => deleteHabit(h.id);

    card.append(chk, emo, info, del);

    list.appendChild(card);

  });

}


function toggleHabit(id) {

  const T = today();

  const h = ST.habits.find(x=>x.id===id);

  const wasDone = h.completedDates.includes(T);

  if (wasDone) {

    h.completedDates = h.completedDates.filter(d=>d!==T);

  } else {

    h.completedDates.push(T);

  }

  h.streak = recalcStreak(h.completedDates);

  if (!wasDone) {

    let bonus = 10;

    if (h.streak===7)  bonus += 50;

    if (h.streak===30) bonus += 200;

    ST.tokens += bonus;

    showToast(`+${bonus} HBT за выполнение!`, 'token');

  }

  save();

  updateTokenDisplay();

  checkBadges();

  renderHabits();

}


function deleteHabit(id) {

  ST.habits = ST.habits.filter(h=>h.id!==id);

  save();

  checkBadges();

  renderHabits();

}


function addHabit(name, emoji, color) {

  ST.habits.push({id: Date.now().toString(), name, emoji, color, streak:0, completedDates:[]});

  save();

  checkBadges();

  renderHabits();

}

function checkBadges() {

  const earned = BADGES.filter(b => {

    if (ST.earnedBadges.includes(b.id)) return false;

    const p = b.prog(ST);

    return p.c >= p.m;

  });

  if (earned.length) {

    earned.forEach(b => {

      ST.earnedBadges.push(b.id);

      showToast(`${b.e} NFT разблокирован: ${b.name}!`, 'nft');

    });

    save();

  }

}
